def tool_leggi_wikipedia(input_str: str) -> str:
    """
    Reads a Wikipedia page and returns its text.
    Input should be the URL of the Wikipedia page.
    """
    st = _state()

    
# --- BLOCCO SICUREZZA: STRINGA VUOTA ---
    if not input_str or not input_str.strip():
        # Se ha cercato qualcosa, gli suggeriamo il primo URL utile
        suggerimento = st.available_urls[0] if st.available_urls else "https://en.wikipedia.org/..."
        return f"CRITICAL ERROR: You provided an empty input. You MUST provide an Action Input like this:\n{suggerimento}"

    # BLOCCO VEGGENTE: Se non ha ancora cercato nulla, la lista è vuota.
    if not st.available_urls:
        return "CRITICAL ERROR: You cannot read a page yet! You MUST call 'tool_cerca_wikipedia' FIRST to get the list of valid URLs."

    try:
        # --- AUTO-CORREZIONE DEL FORMATO ---
        # Se l'agente è stato bravo e ha usato "|| QUESTION:", lo separiamo
        if "||" in input_str:
            parts    = input_str.split("||", 1)
            url_doc  = parts[0].replace("URL:", "").strip()
            question = parts[1].replace("QUESTION:", "").strip()
        else:
            # Se l'agente ha passato solo l'URL, lo perdoniamo e peschiamo la domanda dalla memoria!
            url_doc  = input_str.replace("URL:", "").strip()
            question = st.current_question
            
        url = str(url_doc).replace('"','').replace("'","")

        # Scudo anti-immagine
        if any(x in url.lower() for x in (".jpg", ".jpeg", ".png", "/work/")):
            avail = st.available_urls
            hint  = f"\nUse this URL instead: {avail[0]}" if avail else ""
            return f"CRITICAL ERROR: You passed an image path, not a URL.{hint}"

        # Anti-loop: URL già letto
        if url in st.url_visitati:
            remaining = [u for u in st.available_urls if u not in st.url_visitati]
            if remaining:
                return (
                    f"⚠️ SYSTEM NOTE: You ALREADY READ '{url}'.\n"
                    f"You MUST call 'tool_leggi_wikipedia' with this NEW URL:\n{remaining[0]}"
                )
            return "⚠️ SYSTEM NOTE: All available URLs already read. Output 'Final Answer:' now."

        # URL non in knowledge base
        if url not in wiki_data:
            avail = [u for u in st.available_urls if u not in st.url_visitati]
            hint  = f"\nUse this URL instead: {avail[0]}" if avail else ""
            return f"CRITICAL ERROR: URL not in knowledge base.{hint}"

        # Logging URL letto
        st.url_visitati.append(url)
        if url not in st.read_urls:
            st.read_urls.append(url)

        page = wiki_data[url]

        # Estrazione testo (ignora sezioni boilerplate)
        SKIP_SECTIONS = {"see also","references","external links","further reading","bibliography","notes"}
        sezioni = []
        for title, text in zip(page.get("section_titles", []), page.get("section_texts", [])):
            if title.lower() in SKIP_SECTIONS:
                continue
            sezioni.append(f"--- {title} ---\n{text}")

        tutto = "\n\n".join(sezioni)
        print(f"[READ] {url}")

        limit = 15000
        if len(tutto) > limit:
            tutto = tutto[:limit] + "\n\n...(TEXT TRUNCATED)..."

        # Footer con prossimo URL disponibile
        remaining = [u for u in st.available_urls if u not in st.url_visitati]
        footer = ""
        if remaining:
            footer = (
                f"\n\n[SYSTEM: If this page does NOT contain the exact answer, "
                f"you MUST call 'tool_leggi_wikipedia' with the next URL:\n{remaining[0]}]"
            )

        return tutto + footer

    except Exception as e:
        traceback.print_exc()
        return f"Error reading page: {e}"

def run_agentic_rag(image_path, question, question_id=None,
                    ground_truth="", question_type="unknown",
                    expected_sources=None):

    reset_query_state()
    set_current_question(question)
    vero_qwen.current_image_path = image_path

    start  = now_seconds()
    result = {}
    answer = ""
    error  = None

    try:
        comando = (
            f"[IMG]{image_path}[/IMG]\n"
            f"Question: {question}\n"
            f"Image Path: {image_path}\n"
        )
        result = esecutore.invoke({"input": comando})
        answer = result.get("output", "")

        if not answer or "Agent stopped" in answer or "iteration limit" in answer:
            answer = _fallback_answer(
                image_path, question,
                result.get("intermediate_steps", [])
            )

        answer = _clean_answer(answer)

    except Exception as e:
        error  = str(e)
        answer = _fallback_answer(image_path, question,
                                  result.get("intermediate_steps", []))

    # [A-FIX-1] Leggi URL dallo stato thread-local (sempre aggiornati)
    logged_urls = get_logged_urls()
    read_urls   = get_read_urls()

    step_info   = _summarize_steps(
        result.get("intermediate_steps", []),
        logged_urls=logged_urls,
        read_urls=read_urls,
    )
    ret_metrics = retrieval_metrics(
        step_info.get("read_urls", []),
        expected_sources or [],
        k=args.top_k,
    )

    return build_common_record(
        question_id=question_id,
        model_name="agentic_rag_v4",
        image_path=image_path,
        question=question,
        ground_truth=ground_truth,
        answer=answer,
        question_type=question_type,
        latency_seconds=elapsed(start),
        error=error,
        extra={
            "num_steps": step_info["num_tool_calls"],
            **step_info,
            **ret_metrics,
        },
    )
